=== ADIM 1: demo.ps1 ===

Add-Type @"
using System;
using System.Runtime.InteropServices;
public class WinAPI {
    [DllImport("user32.dll")] public static extern IntPtr GetForegroundWindow();
    [DllImport("user32.dll")] public static extern bool PrintWindow(IntPtr hWnd, IntPtr hdcBlt, uint nFlags);
}
"@
Add-Type -AssemblyName System.Drawing

$app = "C:\Users\alim\Desktop\DNSDegistiriciApp\DNSDegistirici.exe"
$folder = "C:\Users\alim\Desktop\screenshots"
New-Item -ItemType Directory -Path $folder -Force | Out-Null

function Take-Screenshot($name, $delay) {
    Start-Sleep -Seconds $delay
    $hWnd = [WinAPI]::GetForegroundWindow()
    $bmp = New-Object System.Drawing.Bitmap 800, 700
    $g = [System.Drawing.Graphics]::FromImage($bmp)
    $g.CopyFromScreen(0, 0, 0, 0, $bmp.Size)
    $path = Join-Path $folder "$name.png"
    $bmp.Save($path, [System.Drawing.Imaging.ImageFormat]::Png)
    $g.Dispose()
    $bmp.Dispose()
    Write-Host "Captured: $path"
}

Write-Host "Starting demo..."
$p = Start-Process -FilePath $app -PassThru
Start-Sleep -Seconds 3

Take-Screenshot "screen_1" 1
Take-Screenshot "screen_2" 6
Take-Screenshot "screen_3" 6
Take-Screenshot "screen_4" 6
Take-Screenshot "screen_5" 5
